# REQUIRES PYTHON 3.9-3.12 (3.11 is what CI and Streamlit Cloud use).
# On Python 3.13+ this file cannot be installed at all: tensorflow is pinned to
# 2.18.1 below, which only publishes cp39-cp312 wheels, and pip reports the
# unhelpful "No matching distribution found for tensorflow==2.18.1" rather than
# naming the Python version as the cause. If you hit that, check your
# interpreter version first.
#
# Everything except the TF MLP works fine on 3.13 - drop the tensorflow and
# protobuf pins and the tree models still train.
#
# TENSORFLOW PIN - READ BEFORE CHANGING
# hopsworks pins protobuf<5.0.0 (every published version does, up to 5.0.4).
# TensorFlow >=2.19 requires protobuf>=6.31.1. Those ranges do not intersect, so
# `pip install hopsworks tensorflow` with an unpinned TF produces either a
# resolver failure or a silently broken TensorFlow. TF 2.18.x allows
# protobuf>=3.20.3,<6.0.0dev, which overlaps hopsworks at protobuf 4.25.x - so
# 2.18.1 is the newest TensorFlow that can coexist with the feature store.
# If you bump TensorFlow, you must check this constraint again.
tensorflow==2.18.1
protobuf>=4.25.4,<5.0.0

# The [python] extra is REQUIRED, not optional. Plain `hopsworks` installs the
# read path only: reads go over Arrow Flight, but feature-group WRITES go
# through Kafka, and hsfs gates that path on HAS_CONFLUENT_KAFKA. confluent-kafka
# ships solely in this extra, so without it fg.insert() raises ModuleNotFoundError
# while reads keep working - which looks like "the pipeline runs fine but the
# feature store never advances". Also pulls pyarrow/fastavro/httpx.
hopsworks[python]>=4.0,<6.0
